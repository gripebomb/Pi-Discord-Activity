interface ExtensionAPI {

  on(event: 'model_select', handler: (data: ModelSelectEvent) => void): void;
  on(event: 'agent_start', handler: (data: AgentEvent) => void): void;
  on(event: 'agent_end', handler: (data: AgentEvent) => void): void;
  on(event: 'tool_execution_start', handler: (data: ToolExecutionEvent) => void): void;
  on(event: 'tool_execution_end', handler: (data: ToolExecutionEvent) => void): void;
  getSessionInfo(): SessionInfo;
}

interface SessionStartEvent {
  sessionId: string;
  provider: string;
  model: string;
  startedAt: number;

interface AgentEvent {

interface ToolExecutionEvent {

  toolArguments?: Record<string, unknown>;

}

}

<action>
Replace the placeholder handlers with a factory function that subscribes to Pi events:

1. Remove placeholder functions: `onSessionStart`, `onModelChange`, `onThinking`, `onToolCall`, `onFileEdit`, `onIdle`, `onError`

2. Add the factory export at the bottom of the file:
```typescript
export default function (pi: ExtensionAPI) {
  const state = new PresenceState();
  let lastState: string | null = null;

  pi.on('session_start', (event) => {
    state.update({
      sessionId: event.sessionId,
      provider: event.provider ?? 'unknown',
      model: event.model ?? 'unknown',
      state: 'starting'
    });
    lastState = 'starting';
    void publishPresence(state.snapshot());
  });

  pi.on('model_select', (event) => {
    state.update({
      provider: event.provider ?? 'unknown',
      model: event.model ?? 'unknown'
    });
    void publishPresence(state.snapshot());
  });

  pi.on('agent_start', () => {
    if (lastState !== 'thinking') {
      lastState = 'thinking';
      void publishPresence(state.setActivity('thinking'));
    }
  });

  pi.on('tool_execution_start', () => {
    if (lastState !== 'tooling') {
      lastState = 'tooling';
      void publishPresence(state.setActivity('tooling'));
    }
  });

  pi.on('tool_execution_end', () => {
    // Don't change to idle here - wait for agent_end
    // Just acknowledge the tool finished
  });

  pi.on('agent_end', () => {
    lastState = 'idle';
    void publishPresence(state.setActivity('idle'));
  });

  pi.on('session_shutdown', () => {
    lastState = 'idle';
    void publishPresence(state.setActivity('idle'));
  });

  return {
    onUnload: () => {
      console.log('[pi-discord-presence] Extension unloaded');
    }
  };
}
```

3. Keep the imports for PresenceState and publishPresence at the top of the file
</action>
