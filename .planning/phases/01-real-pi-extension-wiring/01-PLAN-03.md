### Task 02: Add timeout to HTTP publish

**Objective:** Prevent blocking when helper is slow or unavailable

<read_first>
- `src/extension/transport.ts` - current implementation
- `src/shared/config.ts` - serverHost, serverPort
</read_first>

<action>
Update `publishPresence()` to include a timeout:

```typescript
export async function publishPresence(payload: PresencePayload): Promise<void> {
  const parsed = presencePayloadSchema.parse(payload);
  const url = `http://${defaultConfig.serverHost}:${defaultConfig.serverPort}/presence`;

  const controller = new AbortController();
  const timeoutId = setTimeout(() => controller.abort(), 2000); // 2 second timeout

  try {
    const response = await fetch(url, {
      method: 'POST',
      headers: {
        'content-type': 'application/json'
      },
      body: JSON.stringify(parsed),
      signal: controller.signal
    });

    clearTimeout(timeoutId);

    if (!response.ok) {
      throw new Error(`Presence publish failed: ${response.status} ${response.statusText}`);
    }
  } catch (error) {
    clearTimeout(timeoutId);
    // Re-throw for caller to handle (caller wraps in try-catch)
    throw error;
  }
}
```

The 2-second timeout ensures presence issues never block Pi event processing.
</action>

<acceptance_criteria>
- [ ] fetch() call has AbortController with 2s timeout
- [ ] Timeout clears on successful response
- [ ] Timeout clears on error (before re-throw)
- [ ] TypeScript compiles without errors
</acceptance_criteria>

```typescript
// scripts/test-helper.ts
// Simple HTTP server that logs presence payloads (for local testing)

import { createServer } from 'http';
import { presencePayloadSchema } from '../src/shared/types.js';

const PORT = 3000;

const server = createServer(async (req, res) => {
  if (req.url === '/presence' && req.method === 'POST') {
    try {
      let body = '';
      req.on('data', chunk => body += chunk);
      req.on('end', () => {
        try {
          const data = JSON.parse(body);
          const parsed = presencePayloadSchema.parse(data);
          
          console.log('\n=== Presence Update ===');
          console.log(`Provider: ${parsed.provider}`);
          console.log(`Model: ${parsed.model}`);
          console.log(`State: ${parsed.state}`);
          console.log(`Session: ${parsed.sessionId ?? 'N/A'}`);
          console.log('========================\n');
          
          res.writeHead(200, { 'content-type': 'application/json' });
          res.end(JSON.stringify({ success: true }));
        } catch (e) {
          console.error('Invalid payload:', e);
          res.writeHead(400, { 'content-type': 'application/json' });
          res.end(JSON.stringify({ error: 'Invalid payload' }));
        }
      });
    } catch (e) {
      res.writeHead(500);
      res.end();
    }
  } else {
    res.writeHead(404);
    res.end();
  }
});

server.listen(PORT, () => {
  console.log(`Test helper listening on http://localhost:${PORT}`);
  console.log('Press Ctrl+C to stop');
});
```
