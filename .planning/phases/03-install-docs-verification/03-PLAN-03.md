This guide walks you through verifying that your pi-discord-presence installation is working correctly end-to-end.

### Phase 1: Helper Daemon Verification

**1.1 Check Helper is Running**
```bash
# Check if helper process is running
ps aux | grep "pi-discord-presence" | grep -v grep

# OR check via service (if set up)
# macOS:
launchctl list | grep pi.discord.presence

# Linux:
systemctl --user status pi-discord-presence.service

# Windows:
nssm status PiDiscordPresence
```

**Expected Output**: Should see the helper process running.

**1.2 Check Helper Logs**
```bash
# macOS/Linux:
tail -f /tmp/pi-discord-presence.log

# Windows:
Get-Content "$env:TEMP\pi-discord-presence\stdout.log" -Wait
```

**Expected Output**: Should see helper starting and Discord connection messages:
```
Pi Discord Presence Helper starting
Discord RPC client connected
```

**1.3 Check Helper HTTP Endpoint**
```bash
# Test helper HTTP endpoint
curl -X POST http://localhost:42666/presence \
  -H "Content-Type: application/json" \
  -d '{
    "app": "pi-coding-agent",
    "provider": "test",
    "model": "test-model",
    "state": "idle",
    "projectName": "test-project",
    "startedAt": 1234567890,
    "sessionId": "test-session",
    "privacyMode": true
  }'
```

**Expected Output**: Should receive HTTP 200 response. Helper may log payload received.

### Troubleshooting

**Discord presence not updating:**
1. Check helper is running and has no errors
2. Verify Discord Client ID is correct
3. Check Discord app has Rich Presence enabled
4. Ensure Discord desktop app is running (not web version)

**Extension not loading:**
1. Verify extension file exists in Pi extension directory
2. Check file permissions
3. Look for errors in Pi agent logs
4. Ensure TypeScript was built correctly

**Helper won't start:**
1. Check Node.js version (20+ required)
2. Verify dependencies are installed
3. Check port 42666 is not in use
4. Look for errors in helper logs

**Privacy not working:**
1. Verify environment variables are set correctly
2. Ensure helper is restarted after changing variables
3. Check helper logs for privacy mode settings

    if echo "$result" | grep -q "$expected_status"; then

echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"

# pi-discord-presence Windows Verification Script
# Automated verification of installation and functionality

param (
    [switch]$Verbose = $false
)

# Set console colors
$Host.UI.RawUI.BackgroundColor = "Black"
$Host.UI.RawUI.ForegroundColor = "White"

Write-Host "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━" -ForegroundColor Cyan

# Track results
$script:totalTests = 0
$script:passedTests = 0
$script:failedTests = 0

# Function to write test result
function Test-Result {
    param (
        [string]$testName,
        [bool]$passed,
        [string]$details = ""
    )
    
    $script:totalTests++
    
    if ($passed) {
        Write-Host "  ✓ $testName" -ForegroundColor Green
        $script:passedTests++
    } else {
        Write-Host "  ✗ $testName" -ForegroundColor Red
        if ($details) {
            Write-Host "    $details" -ForegroundColor Yellow
        }
        $script:failedTests++
    }
}

# Function to check file exists
function Test-FileExists {
    param (
        [string]$testName,
        [string]$filePath
    )
    
    Write-Host "Testing: $testName" -ForegroundColor Blue
    
    if (Test-Path $filePath -PathType Leaf) {
        Test-Result -testName $testName -passed $true
    } else {
        Test-Result -testName $testName -passed $false -details "File not found: $filePath"
    }
    
    Write-Host ""
}

# Function to check directory exists
function Test-DirectoryExists {
    param (
        [string]$testName,
        [string]$dirPath
    )
    
    Write-Host "Testing: $testName" -ForegroundColor Blue
    
    if (Test-Path $dirPath -PathType Container) {
        Test-Result -testName $testName -passed $true
    } else {
        Test-Result -testName $testName -passed $false -details "Directory not found: $dirPath"
    }
    
    Write-Host ""
}

# Function to check process running
function Test-ProcessRunning {
    param (
        [string]$testName,
        [string]$processName
    )
    
    Write-Host "Testing: $testName" -ForegroundColor Blue
    
    $process = Get-Process $processName -ErrorAction SilentlyContinue
    if ($process) {
        Test-Result -testName $testName -passed $true
        $process | Stop-Process -Force -ErrorAction SilentlyContinue
    } else {
        Test-Result -testName $testName -passed $false -details "Process not running: $processName"
    }
    
    Write-Host ""
}

# Function to check service
function Test-ServiceExists {
    param (
        [string]$testName,
        [string]$serviceName
    )
    
    Write-Host "Testing: $testName" -ForegroundColor Blue
    
    try {
        $service = Get-Service -Name $serviceName -ErrorAction Stop
        Test-Result -testName $testName -passed $true
    } catch {
        Test-Result -testName $testName -passed $false -details "Service not found: $serviceName"
    }
    
    Write-Host ""
}

# Get script directory
$scriptDir = Split-Path -Parent $MyInvocation.MyCommand.Definition
$projectDir = Split-Path -Parent $scriptDir

Write-Host "Project directory: $projectDir" -ForegroundColor White
Write-Host ""

# Phase 1: Basic file checks
Write-Host "Phase 1: Basic file checks" -ForegroundColor Cyan
Write-Host "─────────────────────────" -ForegroundColor Cyan

Test-FileExists -testName "Package.json exists" -filePath "$projectDir\package.json"
Test-FileExists -testName "Built helper exists" -filePath "$projectDir\dist\cli\run-helper.js"
Test-FileExists -testName "Built extension exists" -filePath "$projectDir\dist\extension\index.js"

# Test package.json content
Write-Host "Testing: Package.json has pi extension config" -ForegroundColor Blue
try {
    $packageJson = Get-Content "$projectDir\package.json" -Raw | ConvertFrom-Json
    if ($packageJson.pi -and $packageJson.pi.extensions) {
        Test-Result -testName "Package.json has pi extension config" -passed $true
    } else {
        Test-Result -testName "Package.json has pi extension config" -passed $false -details "Missing pi.extensions in package.json"
    }
} catch {
    Test-Result -testName "Package.json has pi extension config" -passed $false -details "Cannot parse package.json: $_"
}

Write-Host "Testing: Package.json has pi-package keyword" -ForegroundColor Blue
try {
    if ($packageJson.keywords -contains "pi-package") {
        Test-Result -testName "Package.json has pi-package keyword" -passed $true
    } else {
        Test-Result -testName "Package.json has pi-package keyword" -passed $false -details "Missing pi-package keyword"
    }
} catch {
    Test-Result -testName "Package.json has pi-package keyword" -passed $false -details "Cannot check keywords: $_"
}

Write-Host ""

# Phase 2: Helper verification
Write-Host "Phase 2: Helper daemon" -ForegroundColor Cyan
Write-Host "────────────────────────" -ForegroundColor Cyan

Test-ProcessRunning -testName "Helper process running" -processName "node"

Write-Host "Testing: Helper HTTP endpoint responds" -ForegroundColor Blue
try {
    $testPayload = @{
