```typescript
import { describe, it, expect, beforeAll, afterAll } from "vitest";
import { createPresenceServer } from "../../src/helper/server.js";
import { publishPresence } from "../../src/extension/transport.js";
import { defaultConfig } from "../../src/shared/config.js";

// Note: This test requires a running Discord RPC
// Use skip on CI or when Discord is not available

describe("End-to-end presence flow", () => {
  let receivedPayloads: any[] = [];
  let server: ReturnType<typeof createPresenceServer>;

  beforeAll(() => {
    server = createPresenceServer(async (payload) => {
      receivedPayloads.push(payload);
    });
  });

  afterAll(() => {
    server.close();
  });

  it("should receive presence payload from extension", async () => {
    const testPayload = {
      app: "pi-coding-agent" as const,
      provider: "anthropic",
      model: "claude-3-5-sonnet",
      state: "thinking" as const,
      startedAt: Math.floor(Date.now() / 1000),
      privacyMode: true
    };

    await publishPresence(testPayload);
    
    // Give server time to process
    await new Promise(r => setTimeout(r, 100));
    
    expect(receivedPayloads.length).toBeGreaterThan(0);
    expect(receivedPayloads[0].model).toBe("claude-3-5-sonnet");
  });
});
```
</action>
